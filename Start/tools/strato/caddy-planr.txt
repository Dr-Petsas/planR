
# PlanR: Startseite, Anmeldung und API über den PlanR-Dienst, die Planer als fertige Dateien
planr.pickadoc.de {
    encode gzip
    header {
        Strict-Transport-Security "max-age=31536000"
        X-Content-Type-Options nosniff
        Referrer-Policy strict-origin-when-cross-origin
        -Server
    }
    handle /planer/* {
        root * /srv/planr
        file_server
    }
    handle {
        reverse_proxy 127.0.0.1:5189
    }
}
